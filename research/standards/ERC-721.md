# ERC-721: Node Operator Identity NFTs

## Overview
ERC-721 NFTs can be used to represent node operator identities in the Olla protocol, providing transparent operator information, reputation tracking, and transferable validator rights. This is a nice-to-have feature for enhanced operator management.

## Status
**Implementation**: Post-V1 nice-to-have  
**Priority**: Low - Enhancement feature  
**Purpose**: Operator transparency and identity management

## Operator NFT Implementation

### Core NFT Contract
```solidity
contract OllaOperatorNFT is ERC721, ERC721Enumerable, AccessControl {
    bytes32 public constant MINTER_ROLE = keccak256("MINTER_ROLE");
    bytes32 public constant OPERATOR_REGISTRY_ROLE = keccak256("OPERATOR_REGISTRY_ROLE");
    
    struct OperatorData {
        string name;                    // Operator name
        string description;            // Operator description
        string imageURI;              // Operator logo/image
        uint256 registrationDate;     // When operator was registered
        uint256 totalStakeManaged;    // Lifetime stake managed
        uint256 totalRewardsGenerated; // Lifetime rewards generated
        uint256 slashingEvents;       // Number of slashing incidents
        uint256 uptimePercentage;     // Current uptime (basis points)
        string[] attributes;          // Additional metadata
    }
    
    mapping(uint256 => OperatorData) public operatorData;
    mapping(address => uint256) public operatorToTokenId;
    uint256 private _nextTokenId = 1;
    
    constructor() ERC721("Olla Operator", "OLLA-OP") {
        _grantRole(DEFAULT_ADMIN_ROLE, msg.sender);
    }
    
    function mintOperatorNFT(
        address operator,
        string memory name,
        string memory description,
        string memory imageURI
    ) external onlyRole(MINTER_ROLE) returns (uint256) {
        require(operatorToTokenId[operator] == 0, "Operator already has NFT");
        
        uint256 tokenId = _nextTokenId++;
        operatorToTokenId[operator] = tokenId;
        
        operatorData[tokenId] = OperatorData({
            name: name,
            description: description,
            imageURI: imageURI,
            registrationDate: block.timestamp,
            totalStakeManaged: 0,
            totalRewardsGenerated: 0,
            slashingEvents: 0,
            uptimePercentage: 10000, // 100% initial
            attributes: new string[](0)
        });
        
        _mint(operator, tokenId);
        return tokenId;
    }
}
```

### Dynamic Metadata Updates
```solidity
contract OperatorMetadataUpdater {
    IOllaOperatorNFT public operatorNFT;
    IPerformanceOracle public performanceOracle;
    
    function updateOperatorMetrics(uint256 tokenId) external {
        address operator = operatorNFT.ownerOf(tokenId);
        
        // Get latest performance data
        (
            uint256 totalStake,
            uint256 totalRewards,
            uint256 uptime,
            uint256 slashing
        ) = performanceOracle.getOperatorMetrics(operator);
        
        // Update NFT metadata
        operatorNFT.updateOperatorData(
            tokenId,
            totalStake,
            totalRewards,
            uptime,
            slashing
        );
        
        emit MetadataUpdated(tokenId, operator);
    }
    
    function batchUpdateMetrics(uint256[] memory tokenIds) external {
        for (uint i = 0; i < tokenIds.length; i++) {
            updateOperatorMetrics(tokenIds[i]);
        }
    }
}
```

### Dynamic Metadata Generation
```solidity
contract OperatorMetadataRenderer {
    string public baseURI = "https://metadata.olla.network/operator/";
    
    function tokenURI(uint256 tokenId) public view override returns (string memory) {
        require(_exists(tokenId), "Token does not exist");
        
        // Generate dynamic metadata URL
        return string(abi.encodePacked(baseURI, tokenId.toString()));
    }
    
    function generateMetadata(uint256 tokenId) external view returns (string memory) {
        OperatorData memory data = operatorData[tokenId];
        address operator = ownerOf(tokenId);
        
        // Get real-time performance metrics
        (uint256 currentStake, uint256 uptime, uint256 rewards) = 
            performanceOracle.getCurrentMetrics(operator);
        
        // Build JSON metadata
        return string(abi.encodePacked(
            '{"name":"', data.name, '",',
            '"description":"', data.description, '",',
            '"image":"', data.imageURI, '",',
            '"attributes":[',
            _buildAttributes(tokenId, data, currentStake, uptime, rewards),
            ']}'
        ));
    }
}
```

## Operator Rights and Transferability

### Transferable Validator Rights
```solidity
contract TransferableOperatorRights {
    mapping(uint256 => bool) public transferRestricted;
    mapping(uint256 => uint256) public transferCooldown;
    uint256 public constant COOLDOWN_PERIOD = 30 days;
    
    modifier transferAllowed(uint256 tokenId) {
        require(!transferRestricted[tokenId], "Transfer restricted");
        require(
            block.timestamp >= transferCooldown[tokenId] + COOLDOWN_PERIOD,
            "Cooldown active"
        );
        _;
    }
    
    function _beforeTokenTransfer(
        address from,
        address to,
        uint256 tokenId,
        uint256 batchSize
    ) internal override transferAllowed(tokenId) {
        super._beforeTokenTransfer(from, to, tokenId, batchSize);
        
        if (from != address(0) && to != address(0)) {
            // Update operator registry with new owner
            operatorRegistry.transferOperator(tokenId, from, to);
            transferCooldown[tokenId] = block.timestamp;
        }
    }
    
    function restrictTransfer(uint256 tokenId) external onlyRole(OPERATOR_REGISTRY_ROLE) {
        transferRestricted[tokenId] = true;
        emit TransferRestricted(tokenId);
    }
}
```

### Stake Migration on Transfer
```solidity
contract OperatorTransferHandler {
    event OperatorTransferred(uint256 indexed tokenId, address indexed from, address indexed to);
    event StakeMigrationStarted(uint256 indexed tokenId, uint256 stakeAmount);
    
    function handleOperatorTransfer(
        uint256 tokenId,
        address from,
        address to
    ) external onlyRole(OPERATOR_NFT_ROLE) {
        
        // Get current stake allocation
        uint256 currentStake = delegationRouter.getOperatorStake(from);
        
        if (currentStake > 0) {
            // Initiate stake migration process
            delegationRouter.initiateOperatorTransfer(from, to, currentStake);
            emit StakeMigrationStarted(tokenId, currentStake);
        }
        
        // Update operator registry
        operatorRegistry.updateOperatorAddress(tokenId, to);
        
        // Transfer collateral
        _transferCollateral(from, to);
        
        emit OperatorTransferred(tokenId, from, to);
    }
}
```

## Reputation and Achievement System

### Achievement Badges
```solidity
contract OperatorAchievements {
    enum Achievement {
        FIRST_MONTH,          // Operated for 1 month
        PERFECT_UPTIME,       // 100% uptime for 6 months  
        LARGE_OPERATOR,       // Managing >1M in stake
        VETERAN_OPERATOR,     // Operating for >1 year
        ZERO_SLASHING,        // No slashing events ever
        TOP_PERFORMER         // Top 10% performance
    }
    
    mapping(uint256 => mapping(Achievement => bool)) public achievements;
    mapping(uint256 => uint256) public achievementCount;
    
    function awardAchievement(uint256 tokenId, Achievement achievement) 
        external onlyRole(ACHIEVEMENT_ROLE) {
        
        if (!achievements[tokenId][achievement]) {
            achievements[tokenId][achievement] = true;
            achievementCount[tokenId]++;
            
            emit AchievementAwarded(tokenId, achievement);
            
            // Update NFT metadata
            _addAchievementAttribute(tokenId, achievement);
        }
    }
}
```

### Reputation Scoring
```solidity
contract OperatorReputationSystem {
    struct ReputationScore {
        uint256 performanceScore;    // 0-1000
        uint256 reliabilityScore;    // 0-1000  
        uint256 communityScore;      // 0-1000
        uint256 longevityScore;      // 0-1000
        uint256 totalScore;          // Weighted average
    }
    
    mapping(uint256 => ReputationScore) public reputationScores;
    
    function calculateReputation(uint256 tokenId) external view returns (uint256) {
        OperatorData memory data = operatorData[tokenId];
        
        uint256 performance = _calculatePerformanceScore(tokenId);
        uint256 reliability = _calculateReliabilityScore(tokenId);
        uint256 community = _calculateCommunityScore(tokenId);
        uint256 longevity = _calculateLongevityScore(data.registrationDate);
        
        // Weighted average (performance weighted highest)
        uint256 totalScore = (performance * 40 + reliability * 30 + community * 20 + longevity * 10) / 100;
        
        return totalScore;
    }
}
```

## Integration with Operator Registry

### NFT-Registry Synchronization
```solidity
contract OperatorRegistryNFTIntegration {
    IOllaOperatorNFT public operatorNFT;
    IOperatorRegistry public operatorRegistry;
    
    function onOperatorApproved(address operator, uint256 stakeCap) external onlyRegistry {
        // Mint NFT when operator is approved
        uint256 tokenId = operatorNFT.mintOperatorNFT(
            operator,
            "New Olla Operator",
            "Professional Aztec network validator",
            "ipfs://QmDefaultOperatorImage"
        );
        
        emit OperatorNFTMinted(operator, tokenId);
    }
    
    function onOperatorRemoved(address operator) external onlyRegistry {
        uint256 tokenId = operatorNFT.operatorToTokenId(operator);
        
        if (tokenId != 0) {
            // Burn NFT when operator is removed
            operatorNFT.burn(tokenId);
            emit OperatorNFTBurned(operator, tokenId);
        }
    }
    
    function onPerformanceUpdate(address operator, uint256 newUptime, uint256 newRewards) 
        external onlyOracle {
        
        uint256 tokenId = operatorNFT.operatorToTokenId(operator);
        
        if (tokenId != 0) {
            // Update NFT metadata with new performance data
            operatorNFT.updatePerformanceData(tokenId, newUptime, newRewards);
            emit OperatorMetadataUpdated(operator, tokenId);
        }
    }
}
```

## Marketplace and Trading

### Operator NFT Marketplace
```solidity
contract OperatorNFTMarketplace {
    struct Listing {
        uint256 tokenId;
        address seller;
        uint256 price;
        bool active;
        uint256 listingTime;
    }
    
    mapping(uint256 => Listing) public listings;
    mapping(uint256 => bool) public approvedForSale;
    
    function listOperatorNFT(uint256 tokenId, uint256 price) external {
        require(operatorNFT.ownerOf(tokenId) == msg.sender, "Not owner");
        require(approvedForSale[tokenId], "Not approved for sale");
        require(!operatorRegistry.hasActiveStake(tokenId), "Has active stake");
        
        listings[tokenId] = Listing({
            tokenId: tokenId,
            seller: msg.sender,
            price: price,
            active: true,
            listingTime: block.timestamp
        });
        
        emit OperatorListed(tokenId, msg.sender, price);
    }
    
    function purchaseOperatorNFT(uint256 tokenId) external payable {
        Listing storage listing = listings[tokenId];
        require(listing.active, "Not for sale");
        require(msg.value >= listing.price, "Insufficient payment");
        
        address seller = listing.seller;
        uint256 price = listing.price;
        
        // Clear listing
        listing.active = false;
        
        // Transfer NFT
        operatorNFT.transferFrom(seller, msg.sender, tokenId);
        
        // Transfer payment (minus marketplace fee)
        uint256 fee = price * marketplaceFee / 10000;
        payable(seller).transfer(price - fee);
        payable(feeRecipient).transfer(fee);
        
        emit OperatorSold(tokenId, seller, msg.sender, price);
    }
}
```

## Use Cases and Benefits

### Transparency and Trust
- **Public Profiles**: Each operator has a publicly visible NFT with performance metrics
- **Historical Tracking**: Immutable record of operator performance over time
- **Verification**: Easy verification of operator identity and credentials
- **Reputation Building**: Long-term reputation building through on-chain history

### Operator Incentives
- **Status Symbol**: NFTs serve as prestigious proof of validator status
- **Achievement System**: Gamified achievements for performance milestones
- **Transferable Value**: Operators can build and transfer reputation
- **Marketing Tool**: Professional operator profiles for attracting delegators

### Protocol Benefits
- **Enhanced Transparency**: Users can easily research operator track records
- **Competition**: Public metrics encourage healthy competition between operators
- **Professionalization**: NFT system encourages professional operator behavior
- **Community Building**: Shared identity and recognition system for operators
