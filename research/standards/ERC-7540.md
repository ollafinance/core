# ERC-7540: Asynchronous Deposit and Redemption

## Overview
ERC-7540 extends the ERC-4626 vault standard to support asynchronous deposit and redemption operations, enabling fair processing of large requests and handling scenarios where immediate liquidity is not available. In Olla's architecture, the ERC-7540 vault holds the underlying Aztec tokens and mints/burns oAztec (ERC-20) tokens as shares.

## Status
**Implementation**: Core requirement for Phase 2  
**Priority**: Critical - Essential for handling validator unstaking delays  
**Base**: Extension of ERC-4626 vault standard

## Interface Definition
```solidity
interface IERC7540 {
    // Request functions
    function requestDeposit(uint256 assets, address receiver) external returns (uint256 requestId);
    function requestRedeem(uint256 shares, address receiver) external returns (uint256 requestId);
    
    // Claim functions
    function claimDeposit(address receiver) external returns (uint256 shares);
    function claimRedeem(address receiver) external returns (uint256 assets);
    
    // Pending request queries
    function pendingDepositRequest(address owner) external view returns (uint256 assets);
    function pendingRedeemRequest(address owner) external view returns (uint256 shares);
    
    // Processing status
    function canProcessRequest(address owner, bool isDeposit) external view returns (bool);
    function previewClaimDeposit(address owner) external view returns (uint256);
    function previewClaimRedeem(address owner) external view returns (uint256);
    
    // Events
    event DepositRequested(address indexed receiver, uint256 assets, uint256 requestId);
    event RedeemRequested(address indexed receiver, uint256 shares, uint256 requestId);
    event DepositClaimed(address indexed receiver, uint256 assets, uint256 shares);
    event RedeemClaimed(address indexed receiver, uint256 shares, uint256 assets);
}
```

### Deposit Request Flow
```solidity
function requestDeposit(uint256 assets, address receiver) 
    external override returns (uint256 requestId) {
    require(assets > 0, "Invalid amount");
    require(pendingDeposits[receiver].assets == 0, "Pending request exists");
    
    // Transfer assets to contract
    IERC20(asset()).safeTransferFrom(msg.sender, address(this), assets);
    
    // Store pending request
    pendingDeposits[receiver] = PendingDeposit({
        assets: assets,
        timestamp: block.timestamp,
        exchangeRate: _getCurrentExchangeRate(),
        claimed: false
    });
    
    // Add to processing queue
    depositQueue.push(receiver);
    
    emit DepositRequested(receiver, assets, block.timestamp);
    return block.timestamp; // Use timestamp as requestId
}
```

### Batch Processing System
```solidity
function processDepositQueue(uint256 maxRequests) external onlyRole(PROCESSOR_ROLE) {
    uint256 processed = 0;
    uint256 totalAssetsToDelegate = 0;
    
    while (depositQueueIndex < depositQueue.length && processed < maxRequests) {
        address receiver = depositQueue[depositQueueIndex];
        PendingDeposit storage request = pendingDeposits[receiver];
        
        if (!request.claimed && request.assets > 0) {
            totalAssetsToDelegate += request.assets;
            processed++;
            
            emit DepositProcessed(receiver, request.assets);
        }
        
        depositQueueIndex++;
    }
    
    // Delegate accumulated assets to operators
    if (totalAssetsToDelegate > 0) {
        _delegateToOperators(totalAssetsToDelegate);
    }
}
```

### Claim Functions
```solidity
function claimDeposit(address receiver) external override returns (uint256 shares) {
    PendingDeposit storage request = pendingDeposits[receiver];
    require(request.assets > 0, "No pending deposit");
    require(!request.claimed, "Already claimed");
    require(_canProcessDeposit(receiver), "Request not ready");
    
    // Calculate shares based on current exchange rate
    shares = convertToShares(request.assets);
    request.claimed = true;
    
    // Mint shares to receiver
    _mint(receiver, shares);
    
    emit DepositClaimed(receiver, request.assets, shares);
    
    // Clean up storage
    delete pendingDeposits[receiver];
}

function claimRedeem(address receiver) external override returns (uint256 assets) {
    PendingRedeem storage request = pendingRedeems[receiver];
    require(request.shares > 0, "No pending redeem");
    require(!request.claimed, "Already claimed");
    require(_canProcessRedeem(receiver), "Request not ready");
    
    // Calculate assets based on current exchange rate
    assets = convertToAssets(request.shares);
    request.claimed = true;
    
    // Transfer assets to receiver
    IERC20(asset()).safeTransfer(receiver, assets);
    
    emit RedeemClaimed(receiver, request.shares, assets);
    
    // Clean up storage
    delete pendingRedeems[receiver];
}
```

## Withdrawal Buffer Integration

### Buffer Management
```solidity
contract WithdrawalBuffer {
    uint256 public targetBufferRatio = 500; // 5% of total assets
    uint256 public maxBufferRatio = 1000;   // 10% maximum
    uint256 public bufferBalance;
    
    function checkBufferAdequacy() external view returns (bool sufficient, uint256 needed) {
        uint256 totalAssets = vault.totalAssets();
        uint256 targetBuffer = totalAssets * targetBufferRatio / 10000;
        
        if (bufferBalance >= targetBuffer) {
            return (true, 0);
        } else {
            return (false, targetBuffer - bufferBalance);
        }
    }
    
    function processWithdrawalsFromBuffer(uint256 maxAmount) external returns (uint256 processed) {
        uint256 available = Math.min(bufferBalance, maxAmount);
        uint256 totalProcessed = 0;
        
        // Process pending redeems that can be satisfied from buffer
        while (redeemQueueIndex < redeemQueue.length && totalProcessed < available) {
            address receiver = redeemQueue[redeemQueueIndex];
            PendingRedeem storage request = pendingRedeems[receiver];
            
            uint256 assetsNeeded = vault.convertToAssets(request.shares);
            
            if (totalProcessed + assetsNeeded <= available && !request.claimed) {
                request.claimed = true;
                totalProcessed += assetsNeeded;
                
                IERC20(vault.asset()).safeTransfer(receiver, assetsNeeded);
                emit WithdrawalProcessed(receiver, request.shares, assetsNeeded);
            }
            
            redeemQueueIndex++;
        }
        
        bufferBalance -= totalProcessed;
        return totalProcessed;
    }
}
```

### Unstaking Coordination
```solidity
function requestUnstakingForWithdrawals() external onlyRole(REBALANCER_ROLE) {
    uint256 pendingRedeemAssets = _calculatePendingRedeemAssets();
    (bool sufficient, uint256 needed) = withdrawalBuffer.checkBufferAdequacy();
    
    if (!sufficient) {
        uint256 totalNeeded = needed + pendingRedeemAssets;
        
        // Request unstaking from operators
        delegationRouter.requestUnstaking(totalNeeded);
        
        emit UnstakingRequested(totalNeeded, block.timestamp);
    }
}

function processUnstakingCompletion(uint256 unstakedAmount) external onlyRole(OPERATOR_ROLE) {
    // Add unstaked assets to withdrawal buffer
    withdrawalBuffer.addToBuffer(unstakedAmount);
    
    // Process pending withdrawals
    withdrawalBuffer.processWithdrawalsFromBuffer(unstakedAmount);
    
    emit UnstakingCompleted(unstakedAmount, block.timestamp);
}
```

## Integration with Existing Standards

### ERC-4626 Compatibility
```solidity
contract LiquidStakingVault is ERC4626, ERC7540 {
    // Override ERC-4626 functions for async behavior
    function deposit(uint256 assets, address receiver) 
        public override returns (uint256 shares) {
        
        // Check if immediate processing is possible
        if (_canProcessImmediately(assets)) {
            return super.deposit(assets, receiver);
        } else {
            // Fall back to async request
            requestDeposit(assets, receiver);
            return 0; // Shares will be minted when claimed
        }
    }
    
    function withdraw(uint256 assets, address receiver, address owner)
        public override returns (uint256 shares) {
        
        uint256 requiredShares = previewWithdraw(assets);
        
        if (_canProcessImmediately(assets)) {
            return super.withdraw(assets, receiver, owner);
        } else {
            // Fall back to async request
            requestRedeem(requiredShares, receiver);
            return requiredShares;
        }
    }
}
```

## Security Considerations

### Rate Manipulation Protection
```solidity
function _validateExchangeRate(uint256 requestRate, uint256 currentRate) internal view {
    uint256 maxDeviation = 500; // 5% maximum rate change
    uint256 deviation = currentRate > requestRate 
        ? currentRate - requestRate 
        : requestRate - currentRate;
    
    require(
        deviation * 10000 <= requestRate * maxDeviation,
        "Exchange rate deviation too large"
    );
}
```

### Request Timeout Handling
```solidity
function _isRequestExpired(uint256 requestTimestamp) internal view returns (bool) {
    return block.timestamp > requestTimestamp + requestTimeout;
}

function cancelExpiredRequest(address user) external {
    PendingDeposit storage deposit = pendingDeposits[user];
    require(_isRequestExpired(deposit.timestamp), "Request not expired");
    require(!deposit.claimed, "Already claimed");
    
    // Refund assets for expired deposit
    uint256 assets = deposit.assets;
    delete pendingDeposits[user];
    
    IERC20(asset()).safeTransfer(user, assets);
    emit RequestCancelled(user, assets, true); // true for deposit
}
```
