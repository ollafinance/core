# ERC-2612: Permit (Gasless Approvals)

## Overview
EIP-2612 extends ERC-20 with gasless approval functionality, enabling users to approve token spending through cryptographic signatures instead of transactions, significantly improving user experience.

## Status
**Implementation**: Critical for UX in Olla protocol  
**Base**: EIP-712 structured data hashing  
**Priority**: High - Essential for one-click staking

## Interface Definition
```solidity
interface IERC2612 {
    function permit(
        address owner,
        address spender,
        uint256 value,
        uint256 deadline,
        uint8 v,
        bytes32 r,
        bytes32 s
    ) external;
    
    function nonces(address owner) external view returns (uint256);
    
    // EIP-712 domain separator
    function DOMAIN_SEPARATOR() external view returns (bytes32);
}
```

## Technical Implementation

### EIP-712 Structured Data
```solidity
contract OAztecToken is ERC20, ERC20Permit {
    // EIP-712 type hash for permit
    bytes32 public constant PERMIT_TYPEHASH = keccak256(
        "Permit(address owner,address spender,uint256 value,uint256 nonce,uint256 deadline)"
    );
    
    // Domain separator for signature verification
    bytes32 private immutable _DOMAIN_SEPARATOR;
    
    // Nonces for replay protection
    mapping(address => uint256) private _nonces;
    
    constructor() ERC20("Liquid Staked Aztec", "oAZTEC") ERC20Permit("oAZTEC") {
        _DOMAIN_SEPARATOR = _domainSeparatorV4();
    }
    
    function permit(
        address owner,
        address spender,
        uint256 value,
        uint256 deadline,
        uint8 v,
        bytes32 r,
        bytes32 s
    ) public override {
        require(block.timestamp <= deadline, "ERC20Permit: expired deadline");
        
        bytes32 structHash = keccak256(
            abi.encode(PERMIT_TYPEHASH, owner, spender, value, _nonces[owner], deadline)
        );
        
        bytes32 hash = _hashTypedDataV4(structHash);
        address signer = ECDSA.recover(hash, v, r, s);
        require(signer == owner, "ERC20Permit: invalid signature");
        
        _nonces[owner]++;
        _approve(owner, spender, value);
    }
    
    function nonces(address owner) public view override returns (uint256) {
        return _nonces[owner];
    }
}
```

## Benefits for Olla Protocol

### One-Click Staking Experience
```typescript
// Traditional flow (2 transactions)
await oAztec.approve(stakingPool.address, amount);
await stakingPool.deposit(amount, userAddress);

// With ERC-2612 (1 transaction)
const signature = await signPermit(amount, deadline);
await stakingPool.depositWithPermit(
    amount,
    userAddress,
    deadline,
    signature.v,
    signature.r,
    signature.s
);
```

### Gasless Approvals
- **No ETH Required**: Users can approve without holding ETH for gas
- **Meta-Transactions**: Enable gasless interactions through relayers
- **Improved Onboarding**: Reduce barriers for new users
- **Cost Efficiency**: Combine approval + action in single transaction

### Enhanced UX Patterns
```solidity
contract LiquidStakingPool {
```

## Security Considerations

### Signature Validation
```solidity
function _validatePermitSignature(
    address owner,
    address spender,
    uint256 value,
    uint256 deadline,
    uint8 v,
    bytes32 r,
    bytes32 s
) internal view {
    require(block.timestamp <= deadline, "Permit expired");
    require(owner != address(0), "Invalid owner");
    
    bytes32 structHash = keccak256(
        abi.encode(
            PERMIT_TYPEHASH,
            owner,
            spender,
            value,
            nonces[owner],
            deadline
        )
    );
    
    bytes32 hash = _hashTypedDataV4(structHash);
    address recoveredSigner = ECDSA.recover(hash, v, r, s);
    
    require(recoveredSigner == owner, "Invalid signature");
}
```

### Protection Mechanisms
- **Nonce Management**: Prevents replay attacks through incremental nonces
- **Deadline Enforcement**: Time-bounded signatures prevent old signature reuse
- **Domain Separation**: EIP-712 domain separator prevents cross-contract signature reuse
- **Signature Validation**: Cryptographic verification of signature authenticity

### Common Attack Vectors
- **Front-Running**: Malicious actors could submit permit transactions first
- **Signature Malleability**: ECDSA signature malleability (mitigated by OpenZeppelin)
- **Phishing**: Users signing malicious permits (education and UI design critical)
- **Replay Attacks**: Cross-chain replay (domain separator prevents this)

## Frontend Integration

### Signature Generation
```typescript
import { signTypedData_v4 } from 'eth-sig-util';

interface PermitData {
  owner: string;
  spender: string;
  value: string;
  nonce: number;
  deadline: number;
}

async function signPermit(
  signer: ethers.Signer,
  token: Contract,
  permitData: PermitData
): Promise<{ v: number; r: string; s: string }> {
  const domain = {
    name: await token.name(),
    version: '1',
    chainId: await signer.getChainId(),
    verifyingContract: token.address,
  };
  
  const types = {
    Permit: [
      { name: 'owner', type: 'address' },
      { name: 'spender', type: 'address' },
      { name: 'value', type: 'uint256' },
      { name: 'nonce', type: 'uint256' },
      { name: 'deadline', type: 'uint256' },
    ],
  };
  
  const signature = await signer._signTypedData(domain, types, permitData);
  return ethers.utils.splitSignature(signature);
}
```

### React Hook Example
```typescript
export function usePermit(tokenContract: Contract) {
  const { account, library } = useWeb3React();
  
  const signPermit = useCallback(async (
    spender: string,
    amount: BigNumber,
    deadline?: number
  ) => {
    if (!account || !library) throw new Error('Wallet not connected');
    
    const nonce = await tokenContract.nonces(account);
    const permitDeadline = deadline || Math.floor(Date.now() / 1000) + 3600; // 1 hour
    
    const permitData = {
      owner: account,
      spender,
      value: amount.toString(),
      nonce: nonce.toNumber(),
      deadline: permitDeadline,
    };
    
    return await signPermit(library.getSigner(), tokenContract, permitData);
  }, [account, library, tokenContract]);
  
  return { signPermit };
}
```

### Unit Tests
```solidity
contract ERC2612Test {
    function testPermitValid() public {
        uint256 deadline = block.timestamp + 1 hours;
        (uint8 v, bytes32 r, bytes32 s) = signPermit(
            owner, spender, amount, deadline
        );
        
        token.permit(owner, spender, amount, deadline, v, r, s);
        
        assertEq(token.allowance(owner, spender), amount);
        assertEq(token.nonces(owner), 1);
    }
    
    function testPermitExpired() public {
        uint256 deadline = block.timestamp - 1;
        (uint8 v, bytes32 r, bytes32 s) = signPermit(
            owner, spender, amount, deadline
        );
        
        vm.expectRevert("ERC20Permit: expired deadline");
        token.permit(owner, spender, amount, deadline, v, r, s);
    }
    
    function testPermitReplay() public {
        uint256 deadline = block.timestamp + 1 hours;
        (uint8 v, bytes32 r, bytes32 s) = signPermit(
            owner, spender, amount, deadline
        );
        
        token.permit(owner, spender, amount, deadline, v, r, s);
        
        vm.expectRevert("ERC20Permit: invalid signature");
        token.permit(owner, spender, amount, deadline, v, r, s);
    }
}
```
