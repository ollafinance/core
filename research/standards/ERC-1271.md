# ERC-1271: Smart Contract Signatures

## Overview
ERC-1271 enables smart contracts to validate signatures on their behalf, allowing multisig wallets, DAOs, and other smart contract wallets to participate in signature-based protocols like governance and permit functions.

## Status
**Implementation**: Post-V1 (governance-focused)  
**Priority**: Medium - Enhances accessibility for institutional users  
**Phase**: Out of scope for V1; revisit with governance rollout

## Interface Definition
```solidity
interface IERC1271 {
    /**
     * @dev Should return whether the signature provided is valid for the provided hash
     * @param hash      Hash of the data to be signed
     * @param signature Signature byte array associated with _hash
     * @return magicValue The bytes4 magic value 0x1626ba7e if valid, 0xffffffff otherwise
     */
    function isValidSignature(bytes32 hash, bytes memory signature) 
        external view returns (bytes4 magicValue);
}

// Magic values
bytes4 constant MAGICVALUE = 0x1626ba7e; // Valid signature
bytes4 constant INVALID_SIGNATURE = 0xffffffff; // Invalid signature
```

## Technical Implementation Details

### Signature Validation Process
1. **Hash Generation**: Create message hash using EIP-712 typed data
2. **Contract Check**: Determine if signer is EOA or smart contract
3. **Validation**: Use ecrecover for EOAs, ERC-1271 for smart contracts
4. **Magic Value**: Return appropriate magic value based on validation result

### Integration with stAztec Token
- **Permit Function**: ERC-2612 permit with ERC-1271 support for smart wallets
- **Transfer Authorization**: Smart contract wallets can authorize transfers
- **DeFi Interactions**: Enables multisig participation in DeFi protocols

## Use Cases in Olla Protocol

### DAO Governance Participation
```solidity
contract OllaGovernor {
    function castVoteBySig(
        uint256 proposalId,
        uint8 support,
        uint8 v,
        bytes32 r,
        bytes32 s
    ) external {
        address voter = ecrecover(
            _hashTypedDataV4(keccak256(abi.encode(
                BALLOT_TYPEHASH,
                proposalId,
                support
            ))),
            v, r, s
        );
        
        // Check if voter is EOA or smart contract
        if (voter.code.length > 0) {
            // Smart contract - use ERC-1271
            require(
                IERC1271(voter).isValidSignature(hash, signature) == MAGICVALUE,
                "Invalid contract signature"
            );
        }
        // EOA signature already verified by ecrecover
        
        _castVote(proposalId, voter, support);
    }
}
```

### Permit for Smart Wallets
```solidity
contract StAztecToken is ERC20, ERC20Permit {
    function permit(
        address owner,
        address spender,
        uint256 value,
        uint256 deadline,
        uint8 v,
        bytes32 r,
        bytes32 s
    ) public override {
        require(block.timestamp <= deadline, "ERC20Permit: expired deadline");
        
        bytes32 structHash = keccak256(
            abi.encode(PERMIT_TYPEHASH, owner, spender, value, _nonces[owner], deadline)
        );
        bytes32 hash = _hashTypedDataV4(structHash);
        
        if (owner.code.length > 0) {
            // Smart contract wallet - use ERC-1271
            bytes memory signature = abi.encodePacked(r, s, v);
            require(
                IERC1271(owner).isValidSignature(hash, signature) == MAGICVALUE,
                "ERC20Permit: invalid signature"
            );
        } else {
            // EOA - use ecrecover
            address signer = ECDSA.recover(hash, v, r, s);
            require(signer == owner, "ERC20Permit: invalid signature");
        }
        
        _nonces[owner]++;
        _approve(owner, spender, value);
    }
}
```

## Smart Contract Wallet Examples

### Multisig Wallet Implementation
```solidity
contract MultisigWallet is IERC1271 {
    uint256 public threshold;
    address[] public owners;
    mapping(address => bool) public isOwner;
    
    function isValidSignature(bytes32 hash, bytes memory signature) 
        external view override returns (bytes4) {
        
        // Decode signature into individual signatures
        uint256 signatureCount = signature.length / 65;
        require(signatureCount >= threshold, "Insufficient signatures");
        
        address[] memory signers = new address[](signatureCount);
        for (uint i = 0; i < signatureCount; i++) {
            bytes32 r;
            bytes32 s;
            uint8 v;
            
            assembly {
                let offset := add(signature, add(0x20, mul(i, 65)))
                r := mload(offset)
                s := mload(add(offset, 32))
                v := byte(0, mload(add(offset, 64)))
            }
            
            address signer = ecrecover(hash, v, r, s);
            require(isOwner[signer], "Invalid signer");
            require(!_contains(signers, signer), "Duplicate signature");
            
            signers[i] = signer;
        }
        
        return MAGICVALUE;
    }
}
```

### DAO Treasury Wallet  
```solidity
contract DAOTreasury is IERC1271 {
    IGovernor public governor;
    mapping(bytes32 => bool) public approvedHashes;
    
    function isValidSignature(bytes32 hash, bytes memory signature) 
        external view override returns (bytes4) {
        
        // Check if hash was approved through governance
        if (approvedHashes[hash]) {
            return MAGICVALUE;
        }
        
        // Check if there's an active proposal that would approve this hash
        uint256 proposalId = uint256(keccak256(signature));
        if (governor.state(proposalId) == IGovernor.ProposalState.Succeeded) {
            return MAGICVALUE;
        }
        
        return INVALID_SIGNATURE;
    }
    
    function approveHash(bytes32 hash) external {
        require(msg.sender == address(governor), "Only governor");
        approvedHashes[hash] = true;
    }
}
```

```solidity
// Example: Account abstraction wallet
contract AAWallet is IERC1271 {
    address public owner;
    mapping(address => bool) public authorizedSigners;
    
    function isValidSignature(bytes32 hash, bytes memory signature) 
        external view override returns (bytes4) {
        
        if (signature.length == 65) {
            // Standard ECDSA signature
            address signer = ECDSA.recover(hash, signature);
            if (signer == owner || authorizedSigners[signer]) {
                return MAGICVALUE;
            }
        } else if (signature.length > 65) {
            // Multi-signature or other scheme
            return _validateComplexSignature(hash, signature);
        }
        
        return INVALID_SIGNATURE;
    }
}
```

### Cross-Chain Governance
```solidity
contract CrossChainGovernance is IERC1271 {
    mapping(uint256 => address) public chainGovernors; // chainId => governor
    
    function isValidSignature(bytes32 hash, bytes memory signature) 
        external view override returns (bytes4) {
        
        // Decode chain ID and signature from signature data
        (uint256 chainId, bytes memory actualSignature) = abi.decode(
            signature, (uint256, bytes)
        );
        
        address chainGovernor = chainGovernors[chainId];
        require(chainGovernor != address(0), "Unsupported chain");
        
        // Verify signature came from authorized governance on other chain
        if (_verifyChainGovernance(chainId, chainGovernor, hash, actualSignature)) {
            return MAGICVALUE;
        }
        
        return INVALID_SIGNATURE;
    }
}
```

## Implementation Patterns

### Signature Validation Library
```solidity
library SignatureValidator {
    function isValidSignature(
        address signer,
        bytes32 hash,
        bytes memory signature
    ) internal view returns (bool) {
        if (signer.code.length > 0) {
            // Smart contract - use ERC-1271
            try IERC1271(signer).isValidSignature(hash, signature) returns (bytes4 magicValue) {
                return magicValue == IERC1271.isValidSignature.selector;
            } catch {
                return false;
            }
        } else {
            // EOA - use ECDSA
            return ECDSA.recover(hash, signature) == signer;
        }
    }
}
```

### Universal Signature Verification
```solidity
contract UniversalSigVerifier {
    using SignatureValidator for address;
    
    function verifySignature(
        address signer,
        bytes32 message,
        bytes memory signature
    ) public view returns (bool) {
        bytes32 ethSignedMessageHash = ECDSA.toEthSignedMessageHash(message);
        return signer.isValidSignature(ethSignedMessageHash, signature);
    }
}
```

## Security Considerations

### Signature Validation
- **Replay Protection**: Ensure signatures can't be reused across different contexts
- **Domain Separation**: Use EIP-712 domain separators to prevent cross-contract replay
- **Signature Uniqueness**: Prevent signature malleability and ensure unique signatures
- **Access Control**: Properly validate who can authorize signatures in smart wallets
